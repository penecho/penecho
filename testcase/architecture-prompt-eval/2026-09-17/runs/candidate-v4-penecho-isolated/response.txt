    mcp:      {x:490, y:430, w:280, h:55, label:"MCP 服务", role:"外部 Agent 接入"},
    model:    {x:880, y:230, w:220, h:70, label:"外部模型 API / CLI 子进程", role:"推理后端"},

  };
  var GROUPS = [
    {x:60,  y:120, w:310, h:360, title:"浏览器（用户端）"},
    {x:455, y:120, w:355, h:420, title:"本地 Node 服务（单进程多职责）"}
  ];
  // edges: from, to, side ports, waypoints, label, style
  var EDGES = [
    {from:"canvas", fs:"r", to:"api", ts:"l", fp:[0,-12], tp:[0,-12],
     label:"HTTP POST /api/ai/command", lp:0.5, lo:-10, style:"solid"},

  // ---------- helpers ----------
  function port(n, side, off){
    var cx = n.x + n.w/2, cy = n.y + n.h/2;
    if(side==="l") return [n.x, cy + (off?off[1]:0)];
    if(side==="r") return [n.x + n.w, cy + (off?off[1]:0)];
    if(side==="t") return [cx + (off?off[0]:0), n.y];
    if(side==="b") return [cx + (off?off[0]:0), n.y + n.h];
    return [cx,cy];
  }
  var ARROW = "M0,0 L10,4 L0,8 Z";
  function defs(){
    var m = (id,c)=>'<marker id="'+id+'" markerWidth="10" markerHeight="8" refX="9" refY="4" orient="auto"><path d="'+ARROW+'" fill="'+c+'"/></marker>';
    return "<defs>"+m("aS","#334155")+m("aD","#94a3b8")+"</defs>";
  }
  function esc(s){return s.replace(/&/g,"&amp;").replace(/</g,"&lt;");}

  function edgePath(e){
    var n1 = NODES[e.from];
    var p1;
    if(e.fs==="out"){ p1 = [n1.x + e.of.dx, n1.y + e.of.dy]; }
    else p1 = port(n1, e.fs, e.fp);
    var p2 = e.tpt ? e.tpt.slice() : port(NODES[e.to], e.ts, e.tp);
    var pts = [p1].concat(e.wp||[]).concat([p2]);
    var d = "M"+pts[0][0]+","+pts[0][1];
    for(var i=1;i<pts.length;i++) d += " L"+pts[i][0]+","+pts[i][1];
    return {d:d, pts:pts};
  }

    s += defs();

      eTargetId["g"+i] = true;
    });
    // edges
    EDGES.forEach(function(e){
      var r = edgePath(e);
      var col = e.style==="dashed" ? "#94a3b8" : "#334155";

      // label
      var lp = e.lp, lo = e.lo;
      var segs = r.pts;
      var segIdx = (e.lseg!==undefined)? e.lseg : 0;
      var a = segs[segIdx], b = segs[segIdx+1];
      var lx = a[0] + (b[0]-a[0])*lp, ly = a[1] + (b[1]-a[1])*lp;
      var tx = lx + (e.lv? e.lv.dx : 0), ty = ly + (e.lv? e.lv.dy : (lo-6));
      var anchor = (e.lv && e.lv.dx>0) ? "start" : "middle";
      s += '<text x="'+tx+'" y="'+ty+'" font-size="12" fill="#475569" text-anchor="'+anchor+'" paint-order="stroke" stroke="#ffffff" stroke-width="4">'+esc(e.label)+'</text>';
    });
    // nodes
    for(var id in NODES){
      var n = NODES[id];
      s += '<g><rect x="'+n.x+'" y="'+n.y+'" width="'+n.w+'" height="'+n.h+'" rx="7" fill="#ffffff" stroke="#64748b" stroke-width="1.6"/>';

    }
    return s;
  }
  var eTargetId = {};
